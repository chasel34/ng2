Representative TraceProcessor SQL
=================================

  select count(*), max(dur)
  from actual_frame_timeline_slice
  where layer_name like '%com.chasel.ng2.n%' and ts >= :lo and ts < :hi;

  select s.name, count(*), sum(s.dur), max(s.dur)
  from slice s
  join thread_track tt on tt.id=s.track_id
  join thread t using(utid)
  where t.tid in (760,8652) and s.ts < :hi and s.ts+s.dur > :lo
  group by s.name;

  select state, sum(min(ts+dur,:hi)-max(ts,:lo))
  from thread_state
  where utid=:ui_or_render_utid and ts < :hi and ts+dur > :lo
  group by state;
